    function cambiarSeccion(id) {
      document.querySelectorAll('.seccion').forEach(sec => sec.classList.remove('activa'));
      document.getElementById(id).classList.add('activa');
      if(id === 'panel') cargarDatosDeCalendar();
    }

    async function cargarDatosDeCalendar() {
      const loader = document.getElementById('cargandoEventos');
      loader.classList.remove('hidden');
      
      try {
        const respuesta = await fetch(`${API_URL}?action=obtenerEventos`);
        const res = await respuesta.json();
        
        loader.classList.add('hidden');
        if(res.exito) {
          todosLosEventos = res.eventos;
          aplicarFiltro(vistaActual);
        } else {
            console.error("Error desde el servidor:", res.error);
        }
      } catch (error) {
        loader.classList.add('hidden');
        console.error("Error de conexión:", error);
      }
    }

    async function guardarEvento(e) {
      e.preventDefault();
      const btn = document.getElementById('btnGuardar');
      const originalText = btn.innerText;
      btn.innerHTML = '<i class="fa-solid fa-circle-notch fa-spin"></i>';
      btn.disabled = true;
      
      document.getElementById('errorFormulario').classList.add('hidden');

      const id = document.getElementById('eventoId').value;
      const datos = {
        tipo: document.getElementById('tipoActividad').value,
        titulo: document.getElementById('titulo').value,
        fecha: document.getElementById('fecha').value,
        hora: document.getElementById('hora').value,
        duracion: document.getElementById('duracion').value,
        descripcion: document.getElementById('descripcion').value
      };

      const payload = {
        action: id ? 'actualizar' : 'crear',
        id: id,
        datos: datos
      };

      try {
        const respuesta = await fetch(API_URL, {
            method: 'POST',
            body: JSON.stringify(payload)
        });
        const res = await respuesta.json();
        finalizarGuardado(res, btn, originalText);
      } catch(error) {
          finalizarGuardado({exito: false, error: "Error de conexión"}, btn, originalText);
      }
    }

    async function ejecutarEliminar() {
      if(!eventoAEliminar) return;
      const btn = document.getElementById('btnConfirmarEliminar');
      btn.innerHTML = '<i class="fa-solid fa-circle-notch fa-spin"></i>';
      btn.disabled = true;

      const payload = { action: 'eliminar', id: eventoAEliminar };

      try {
        const respuesta = await fetch(API_URL, {
            method: 'POST',
            body: JSON.stringify(payload)
        });
        const res = await respuesta.json();
        
        btn.innerHTML = 'Sí, eliminar';
        btn.disabled = false;
        cerrarModalEliminar();
        
        if(res.exito) cargarDatosDeCalendar();
        else alert('Error al eliminar: ' + res.error);

      } catch(error) {
          btn.innerHTML = 'Sí, eliminar';
          btn.disabled = false;
          alert("Error de conexión");
      }
    }

    function filtrarVista(vista, btnElement) {
      vistaActual = vista;
      document.querySelectorAll('.tab-btn').forEach(btn => {
        btn.classList.remove('active', 'bg-indigo-600', 'text-white', 'shadow-md');
        btn.classList.add('text-slate-600', 'hover:bg-slate-50');
      });
      btnElement.classList.add('active', 'bg-indigo-600', 'text-white', 'shadow-md');
      btnElement.classList.remove('text-slate-600', 'hover:bg-slate-50');
      aplicarFiltro(vista);
    }

    function aplicarFiltro(vista) {
      const hoy = new Date();
      hoy.setHours(0,0,0,0);
      
      const inicioSemana = new Date(hoy);
      inicioSemana.setDate(hoy.getDate() - hoy.getDay() + 1); 
      const finSemana = new Date(inicioSemana);
      finSemana.setDate(inicioSemana.getDate() + 6); 

      let eventosFiltrados = [];
      let textoTitulo = "";

      if(vista === 'hoy') {
        textoTitulo = "Actividades de Hoy";
        eventosFiltrados = todosLosEventos.filter(e => {
          const d = new Date(e.fechaInicio);
          d.setHours(0,0,0,0);
          return d.getTime() === hoy.getTime();
        });
      } 
      else if(vista === 'semana') {
        textoTitulo = `Semana del ${inicioSemana.getDate()} al ${finSemana.getDate()}`;
        eventosFiltrados = todosLosEventos.filter(e => {
          const d = new Date(e.fechaInicio);
          return d >= inicioSemana && d <= new Date(finSemana.getTime() + 86400000);
        });
      } 
      else if(vista === 'mes') {
        const meses = ['Enero', 'Febrero', 'Marzo', 'Abril', 'Mayo', 'Junio', 'Julio', 'Agosto', 'Septiembre', 'Octubre', 'Noviembre', 'Diciembre'];
        textoTitulo = `Mes de ${meses[hoy.getMonth()]}`;
        eventosFiltrados = todosLosEventos.filter(e => {
          return new Date(e.fechaInicio).getMonth() === hoy.getMonth();
        });
      }

      document.getElementById('tituloRangoFechas').innerText = textoTitulo;
      renderizarLista(eventosFiltrados);
    }

    function renderizarLista(eventos) {
      const contenedor = document.getElementById('listaEventos');
      if (eventos.length === 0) {
        contenedor.innerHTML = `
          <div class="text-center py-12 bg-white rounded-2xl border-2 border-dashed border-slate-200 text-slate-400">
            <i class="fa-solid fa-mug-hot text-4xl mb-3 text-slate-300"></i>
            <p class="font-medium">No hay actividades agendadas en este periodo.</p>
          </div>`;
        return;
      }

      const mesesCortos = ['Ene','Feb','Mar','Abr','May','Jun','Jul','Ago','Sep','Oct','Nov','Dic'];
      
      contenedor.innerHTML = eventos.map(ev => {
        const fechaObj = new Date(ev.fechaInicio);
        const fechaFinObj = new Date(ev.fechaFin);
        const dia = fechaObj.getDate();
        const mesStr = mesesCortos[fechaObj.getMonth()];
        
        const horaInicioStr = fechaObj.toLocaleTimeString([], {hour: '2-digit', minute:'2-digit'});
        const horaFinStr = fechaFinObj.toLocaleTimeString([], {hour: '2-digit', minute:'2-digit'});
        
        const estilo = estiloTipos[ev.tipo] || estiloTipos['Otras'];

        return `
          <div class="group relative flex items-center justify-between p-5 bg-white rounded-2xl border border-slate-200 hover:border-indigo-300 shadow-sm hover:shadow-md transition-all duration-300 transform hover:-translate-y-0.5">
            <div class="absolute left-0 top-0 bottom-0 w-1.5 bg-gradient-to-b ${estilo.gradient} rounded-l-2xl"></div>
            <div class="flex items-center space-x-5 pl-3 w-full">
              <div class="flex flex-col items-center justify-center ${estilo.bg} ${estilo.text} w-16 h-16 rounded-xl border ${estilo.border} flex-shrink-0 shadow-inner">
                <span class="text-[10px] font-bold uppercase tracking-wider">${mesStr}</span>
                <span class="text-xl font-extrabold leading-none mt-0.5">${dia}</span>
              </div>
              <div class="flex-1 min-w-0">
                <div class="flex items-center space-x-2 mb-1">
                  <span class="inline-flex items-center px-2 py-0.5 rounded text-xs font-bold ${estilo.bg} ${estilo.text} border ${estilo.border}">
                    <i class="fa-solid ${estilo.icon} mr-1.5"></i> ${ev.tipo || 'Otras'}
                  </span>
                </div>
                <h4 class="text-lg font-bold text-slate-800 truncate">${ev.titulo}</h4>
                <div class="flex items-center text-sm text-slate-500 mt-1 space-x-4">
                  <span class="flex items-center font-medium"><i class="fa-regular fa-clock text-slate-400 mr-1.5"></i> ${horaInicioStr} - ${horaFinStr}</span>
                  ${ev.descripcion ? `<span class="truncate max-w-[200px] hidden md:inline-block border-l border-slate-300 pl-4">${ev.descripcion}</span>` : ''}
                </div>
              </div>
            </div>
            
            <div class="flex items-center space-x-2 md:opacity-0 group-hover:opacity-100 transition-opacity duration-300 bg-white pl-4">
              <button onclick="prepararEdicion('${ev.id}')" title="Editar" class="w-10 h-10 rounded-full bg-slate-50 hover:bg-amber-100 text-slate-500 hover:text-amber-600 flex items-center justify-center transition shadow-sm"><i class="fa-solid fa-pen text-sm"></i></button>
              <button onclick="abrirModalEliminar('${ev.id}')" title="Eliminar" class="w-10 h-10 rounded-full bg-slate-50 hover:bg-red-100 text-slate-500 hover:text-red-600 flex items-center justify-center transition shadow-sm"><i class="fa-solid fa-trash text-sm"></i></button>
            </div>
          </div>
        `;
      }).join('');
    }

    function prepararEdicion(id) {
      const ev = todosLosEventos.find(e => e.id === id);
      if(!ev) return;
      
      const fechaInicio = new Date(ev.fechaInicio);
      const fechaFin = new Date(ev.fechaFin);
      const duracionMinutos = Math.round((fechaFin - fechaInicio) / 60000);

      const anio = fechaInicio.getFullYear();
      const mes = String(fechaInicio.getMonth() + 1).padStart(2, '0');
      const dia = String(fechaInicio.getDate()).padStart(2, '0');
      const hora = String(fechaInicio.getHours()).padStart(2, '0');
      const minutos = String(fechaInicio.getMinutes()).padStart(2, '0');

      const soloFecha = `${anio}-${mes}-${dia}`;
      const soloHora = `${hora}:${minutos}`;

      document.getElementById('eventoId').value = ev.id;
      
      const tipoSelect = document.getElementById('tipoActividad');
      const existeOpcion = Array.from(tipoSelect.options).some(opt => opt.value === ev.tipo);
      tipoSelect.value = existeOpcion ? ev.tipo : 'Otras';
      
      document.getElementById('titulo').value = ev.titulo;
      document.getElementById('fecha').value = soloFecha;
      document.getElementById('hora').value = soloHora;
      document.getElementById('duracion').value = duracionMinutos;
      document.getElementById('descripcion').value = ev.descripcion || '';
      
      document.getElementById('tituloFormulario').innerText = 'Editar Actividad';
      document.getElementById('btnGuardar').innerText = 'Actualizar Actividad';
      document.getElementById('errorFormulario').classList.add('hidden');
      
      cambiarSeccion('actividades');
    }

    function finalizarGuardado(respuesta, btn, originalText) {
      btn.innerHTML = originalText;
      btn.disabled = false;
      
      if(respuesta.exito) { 
        cambiarSeccion('panel'); 
      } else { 
        document.getElementById('textoErrorFormulario').innerText = respuesta.error;
        document.getElementById('errorFormulario').classList.remove('hidden');
      }
    }

    function abrirModalEliminar(id) {
      eventoAEliminar = id;
      const modal = document.getElementById('modalEliminar');
      const box = document.getElementById('modalBox');
      modal.classList.remove('hidden');
      setTimeout(() => {
        box.classList.remove('scale-95', 'opacity-0');
        box.classList.add('scale-100', 'opacity-100');
      }, 10);
    }

    function cerrarModalEliminar() {
      eventoAEliminar = null;
      const modal = document.getElementById('modalEliminar');
      const box = document.getElementById('modalBox');
      box.classList.remove('scale-100', 'opacity-100');
      box.classList.add('scale-95', 'opacity-0');
      setTimeout(() => modal.classList.add('hidden'), 200);
    }

    function guardarConfiguracion() {
      const nombre = document.getElementById('inputNombreConfig').value;
      if (nombre.trim() !== '') {
        localStorage.setItem('coordinacion_nombre', nombre);
        document.getElementById('nombreUsuario').innerText = nombre;
      }
      const fileInput = document.getElementById('inputLogoConfig');
      if (fileInput.files && fileInput.files[0]) {
        const reader = new FileReader();
        reader.onload = function(e) {
          const src = e.target.result;
          localStorage.setItem('coordinacion_logo', src);
          aplicarLogo(src);
        };
        reader.readAsDataURL(fileInput.files[0]);
      }
      cambiarSeccion('panel');
    }

    function aplicarLogo(src) {
      const imgSidebar = document.getElementById('logoSidebar');
      const iconSidebar = document.getElementById('iconSidebarContainer');
      imgSidebar.src = src;
      imgSidebar.classList.remove('hidden');
      iconSidebar.classList.add('hidden');
      
      const previewImg = document.getElementById('previewLogoConfig');
      const previewIcon = document.getElementById('previewIconConfig');
      previewImg.src = src;
      previewImg.classList.remove('hidden');
      previewIcon.classList.add('hidden');
    }

    window.onload = function() {
      const n = localStorage.getItem('coordinacion_nombre');
      if (n) {
        document.getElementById('nombreUsuario').innerText = n;
        document.getElementById('inputNombreConfig').value = n;
      } else {
        document.getElementById('nombreUsuario').innerText = 'Usuario';
      }
      const l = localStorage.getItem('coordinacion_logo');
      if (l) aplicarLogo(l);
      cargarDatosDeCalendar();
    };
